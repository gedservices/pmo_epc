-- =========================================================
-- REFERENTIELS DE BASE
-- =========================================================

-- -------------------------
-- PHASES EPC
-- -------------------------
create table phases (
    id bigint generated always as identity primary key,
    code varchar(10) not null unique,
    nom varchar(100) not null,
    ordre int not null default 0,
    actif boolean default true
);

-- -------------------------
-- CLASSES DOCUMENTAIRES
-- -------------------------
create table classes_documentaires (
    id bigint generated always as identity primary key,
    numero int not null unique check (numero between 1 and 6),
    libelle varchar(50) not null,
    description text,
    cycle_simplifie boolean default false
    -- cycle_simplifie = true pour classes 4,5,6
);

-- -------------------------
-- STATUTS CYCLE DE VIE
-- -------------------------
create table statuts_cycle_vie (
    id bigint generated always as identity primary key,
    code varchar(10) not null unique,
    libelle varchar(100) not null,
    description text,
    est_terminal boolean default false
);

-- -------------------------
-- SEQUENCES DE STATUTS
-- (phase + classe => sequence ordonnee)
-- -------------------------
create table sequences_statuts (
    id bigint generated always as identity primary key,
    phase_id bigint not null references phases(id),
    classe_num int not null check (classe_num between 1 and 6),
    statut_id bigint not null references statuts_cycle_vie(id),
    ordre int not null,
    est_statut_final boolean default false,
    unique (phase_id, classe_num, ordre)
);

-- -------------------------
-- DISCIPLINES
-- -------------------------
create table disciplines (
    id bigint generated always as identity primary key,
    code varchar(10) not null unique,
    nom varchar(100) not null,
    actif boolean default true
);

-- -------------------------
-- TYPES DE DOCUMENTS TECHNIQUES
-- -------------------------
create table doc_types_technique (
    id bigint generated always as identity primary key,
    code varchar(20) not null unique,
    libelle varchar(100) not null,
    classe_num int references classes_documentaires(numero),
    discipline_id bigint references disciplines(id),
    actif boolean default true
);

-- -------------------------
-- TYPES DE DOCUMENTS CORRESPONDANCE
-- -------------------------
create table doc_types_correspondance (
    id bigint generated always as identity primary key,
    code varchar(20) not null unique,
    libelle varchar(100) not null,
    actif boolean default true
);

-- =========================================================
-- GEOGRAPHIE
-- =========================================================

-- -------------------------
-- PAYS
-- -------------------------
create table pays (
    id bigint generated always as identity primary key,
    code_pays varchar(5) not null unique,

-- -------------------------
-- SITES

-- -------------------------
-- SECTEURS

create table secteurs (

    code_secteur varchar(10) not null unique,

-- -------------------------
-- SYSTEMES
-- -------------------------
create table systemes (
    id bigint generated always as identity primary key,
    code_systeme varchar(20) not null unique,
    nom varchar(100) not null,
    secteur_id bigint not null references secteurs(id),
    actif boolean default true
);

-- -------------------------
-- SOUS-SYSTEMES (UNITES)
-- -------------------------
create table sous_systemes (
    id bigint generated always as identity primary key,
    code_sous_systeme varchar(20) not null unique,

-- TAGS

    id bigint generated always as identity primary key,
    code_tag varchar(50) not null unique,

-- =========================================================
-- ORGANISATION
-- =========================================================

-- -------------------------
-- ENTITES
-- -------------------------
create table entites (
    id bigint generated always as identity primary key,
    nom varchar(100) not null unique,
    type varchar(20) not null default 'INTERNE'
        check (type in ('INTERNE','EXTERNE')),
    description text,
    actif boolean default true
);

-- -------------------------
-- UTILISATEURS
-- -------------------------
create table utilisateurs (
    id bigint generated always as identity primary key,
    nom varchar(100) not null,
    prenom varchar(100) not null,
    email varchar(150) not null unique,
    entite_id bigint references entites(id),
    role varchar(30) not null default 'MEMBER'
        check (role in (
            'ADMIN',
            'PMO',
            'MANAGER',
            'LEAD_DOC_CONTROLLER',
            'DOC_CONTROLLER',
            'LEADER_TECHNIQUE',
            'SPECIALIST',
            'MEMBER',
            'CONTRACTOR'
        )),
    est_contractor boolean default false,
    actif boolean default true,
    date_creation timestamptz default now(),
    date_modification timestamptz default now()
);

-- =========================================================
-- CONTENEURS DE CYCLE DE VIE
-- =========================================================

create table conteneurs (
    id bigint generated always as identity primary key,
    code varchar(20) not null unique,
    libelle varchar(100) not null,
    description text,
    ordre int default 0,
    est_archive boolean default false,
    actif boolean default true
);

-- =========================================================
-- PROJETS
-- =========================================================

create table projets (
    id bigint generated always as identity primary key,
    code varchar(50) not null unique,
    nom varchar(255) not null,
    description text,
    type_projet varchar(50),

    conteneur_id bigint not null references conteneurs(id),

    statut varchar(30) not null default 'Ouvert'
        check (statut in (
            'Ouvert','En cours','En attente',
            'Suspendu','Clôturé','Annulé','Finalisé'
        )),

    priorite varchar(30)
        check (priorite in ('Critique','Haute','Moyenne','Basse')),

    responsable_id bigint references utilisateurs(id),
    entite_id bigint references entites(id),

    -- geographie principale
    pays_id bigint references pays(id),
    site_id bigint references sites(id),
    secteur_id bigint references secteurs(id),

    date_debut_prevue date,
    date_fin_prevue date,
    date_debut_reelle date,
    date_fin_reelle date,

    budget_prevu numeric(15,2) default 0,
    budget_reel numeric(15,2) default 0,

    progression_calculee numeric(5,2) default 0
        check (progression_calculee >= 0 and progression_calculee <= 100),

    poids_portefeuille numeric(5,2) default 0,

    -- EVM
    planned_value numeric(15,2) default 0,
    earned_value numeric(15,2) default 0,
    actual_cost numeric(15,2) default 0,

    est_generique boolean default false,
    est_lecture_seule boolean default false,

    date_creation timestamptz default now(),
    date_modification timestamptz default now(),
    cree_par bigint references utilisateurs(id),
    modifie_par bigint references utilisateurs(id)
);

-- -------------------------
-- PROJET <-> ENTITES (M2M)
-- -------------------------
create table projet_entites (
    projet_id bigint not null references projets(id) on delete cascade,
    entite_id bigint not null references entites(id) on delete cascade,
    role_entite varchar(100),
    primary key (projet_id, entite_id)
);

-- -------------------------
-- PROJET <-> SITES (M2M)
-- -------------------------
create table projet_sites (
    projet_id bigint not null references projets(id) on delete cascade,
    site_id bigint not null references sites(id) on delete cascade,
    primary key (projet_id, site_id)
);

-- -------------------------
-- LOG MIGRATIONS CONTENEURS
-- -------------------------
create table migration_conteneur_log (
    id bigint generated always as identity primary key,
    projet_id bigint not null references projets(id),
    conteneur_source_id bigint not null references conteneurs(id),
    conteneur_cible_id bigint not null references conteneurs(id),
    date_migration timestamptz default now(),
    migre_par bigint references utilisateurs(id),
    commentaire text
);

-- =========================================================
-- COMMANDES / CONTRATS
-- =========================================================

create table commandes (
    id bigint generated always as identity primary key,
    projet_id bigint not null references projets(id) on delete cascade,

    code_commande varchar(50) not null unique,
    code_originator varchar(4) not null,
    nom varchar(255) not null,
    description text,

    contractor varchar(255),
    entite_contractor_id bigint references entites(id),

    phase_id bigint references phases(id),

    statut varchar(30) not null default 'Ouvert'
        check (statut in (
            'Ouvert','En cours','En attente',
            'Suspendu','Terminée','Annulée'
        )),

    priorite varchar(30)
        check (priorite in ('Critique','Haute','Moyenne','Basse')),

    responsable_id bigint references utilisateurs(id),

    -- geographie specifique commande
    pays_id bigint references pays(id),
    site_id bigint references sites(id),
    secteur_id bigint references secteurs(id),

    date_debut_prevue date,
    date_fin_prevue date,
    date_debut_reelle date,
    date_fin_reelle date,

    budget_prevu numeric(15,2) default 0,
    budget_reel numeric(15,2) default 0,

    poids numeric(5,2) default 0
        check (poids >= 0 and poids <= 100),

    progression_calculee numeric(5,2) default 0
        check (progression_calculee >= 0 and progression_calculee <= 100),

    -- durees de revue par defaut pour cette commande
    duree_revue_company_defaut int default 15,
    duree_revue_contractor_defaut int default 15,

    -- EVM commande
    planned_value numeric(15,2) default 0,
    earned_value numeric(15,2) default 0,
    actual_cost numeric(15,2) default 0,

    est_generique boolean default false,

    date_creation timestamptz default now(),
    date_modification timestamptz default now(),
    cree_par bigint references utilisateurs(id),
    modifie_par bigint references utilisateurs(id)
);

-- =========================================================
-- TACHES
-- =========================================================

create table taches (
    id bigint generated always as identity primary key,
    projet_id bigint not null references projets(id) on delete cascade,

            'Suspendue','Annulée'

    poids numeric(5,2) default 0
        check (poids >= 0 and poids <= 100),

    avancement numeric(5,2) default 0
        check (avancement >= 0 and avancement <= 100),

    date_fin_prevue date,
    date_debut_reelle date,
    date_fin_reelle date,

    cout_prevu numeric(15,2) default 0,
    cout_reel numeric(15,2) default 0,

    planned_value numeric(15,2) default 0,
    earned_value numeric(15,2) default 0,

    est_jalon boolean default false,

    date_creation timestamptz default now(),
    date_modification timestamptz default now(),
    cree_par bigint references utilisateurs(id),
    modifie_par bigint references utilisateurs(id)
);

-- =========================================================
-- RISQUES
-- =========================================================

create table risques (
    id bigint generated always as identity primary key,
    projet_id bigint not null references projets(id) on delete cascade,
    commande_id bigint references commandes(id) on delete cascade,

    titre varchar(255) not null,
    description text,

    probabilite int not null check (probabilite between 1 and 5),
    impact int not null check (impact between 1 and 5),
    criticite int generated always as (probabilite * impact) stored,

    statut varchar(30) default 'Ouvert'
        check (statut in ('Ouvert','En cours','Clôturé','Accepté')),

    plan_mitigation text,
    responsable_id bigint references utilisateurs(id),

    date_identification date default current_date,
    date_echeance date,
    date_creation timestamptz default now(),
    date_modification timestamptz default now()
);

-- =========================================================
-- HISTORIQUE AVANCEMENT & EVM SNAPSHOTS
-- =========================================================

create table historique_avancement (
    id bigint generated always as identity primary key,
    projet_id bigint not null references projets(id) on delete cascade,
    commande_id bigint references commandes(id) on delete cascade,

    date_snapshot date not null,
    avancement_reel numeric(5,2),
    avancement_prevu numeric(5,2),

    -- EVM snapshot
    planned_value numeric(15,2),
    earned_value numeric(15,2),
    actual_cost numeric(15,2),
    spi numeric(8,4),
    cpi numeric(8,4)
);

create table evm_snapshots (
    id bigint generated always as identity primary key,
    projet_id bigint not null references projets(id) on delete cascade,
    commande_id bigint references commandes(id) on delete cascade,
    tache_id bigint references taches(id) on delete cascade,

    date_snapshot date not null,

    budget_total numeric(15,2),
    planned_value numeric(15,2),
    earned_value numeric(15,2),
    actual_cost numeric(15,2),

    spi numeric(8,4)
        generated always as (
            case when planned_value > 0
                then earned_value / planned_value
                else null end
        ) stored,

    cpi numeric(8,4)
        generated always as (
            case when actual_cost > 0
                then earned_value / actual_cost
                else null end
        ) stored,

    sv numeric(15,2)
        generated always as (earned_value - planned_value) stored,

    cv numeric(15,2)
        generated always as (earned_value - actual_cost) stored,

    date_creation timestamptz default now()
);

-- -------------------------
-- DOCUMENTS MDR (lignes)
-- -------------------------
create table documents_mdr (
    id bigint generated always as identity primary key,
    mdr_id bigint not null references mdr(id) on delete cascade,
    commande_id bigint not null references commandes(id) on delete cascade,
    projet_id bigint not null references projets(id),

    -- IDENTIFICATION
    numero_document varchar(100) not null unique,
    titre varchar(255) not null,

    -- CLASSIFICATION
    phase_id bigint references phases(id),
    discipline_id bigint references disciplines(id),
    classe_num int not null default 1
        references classes_documentaires(numero),
    doc_type_id bigint references doc_types_technique(id),

    -- GEOGRAPHIE & HIERARCHIE
    pays_id bigint references pays(id),
    site_id bigint references sites(id),
    secteur_id bigint references secteurs(id),
    systeme_id bigint references systemes(id),
    sous_systeme_id bigint references sous_systemes(id),
    tag_id bigint references tags(id),

    -- PARAMETRES DE REVUE
    duree_revue_company int default 15,
    duree_revue_contractor int default 15,

    -- SEQUENCE ET STATUT FINAL ATTENDU
    statut_final_attendu varchar(10),
    sequence_statuts text,
    -- sequence_statuts : stockage JSON ou texte delimiteur
    -- ex : 'IFR,IFA,IFC' calculé à la création

    -- SUIVI COURANT
    revision_courante varchar(5) default '00',
    statut_courant varchar(10),
    code_retour_courant varchar(1)
        check (code_retour_courant in ('A','B','C','D')),

    est_finalise boolean default false,
    date_finalisation date,

    -- DATES PREVISIONNELLES ET REELLES
    date_emission_prevue date,
    date_emission_reelle date,
    date_retour_prevu date,
    date_retour_reel date,
    date_prochaine_emission_prevue date,

    -- FICHIER COURANT
    fichier_natif varchar(500),

    -- METADATA
    commentaire_interne text,
    date_creation timestamptz default now(),
    date_modification timestamptz default now(),
    cree_par bigint references utilisateurs(id),
    modifie_par bigint references utilisateurs(id)
);

-- FK differee tache_documents -> documents_mdr
alter table tache_documents
    add constraint fk_tache_doc_mdr
    foreign key (document_mdr_id)
    references documents_mdr(id) on delete cascade;

        check (code_retour in ('A','B','C','D')),

    date_emission_contractor date,
    date_reception_company date,
    date_retour_company date,
    date_limite_retour date,

    fichier_revision varchar(500),

    transmittal_entrant_id bigint,
    transmittal_sortant_id bigint,
    -- FK vers transmittals ajoutees apres

    commentaires_revue_consolides text,
    est_en_retard boolean default false,

    date_creation timestamptz default now(),
    cree_par bigint references utilisateurs(id),

    unique (document_mdr_id, numero_revision)
);

-- =========================================================
-- TRANSMITTALS

create table transmittals (
    id bigint generated always as identity primary key,
    numero_transmittal varchar(50) not null unique,

    type varchar(20) not null
        check (type in ('ENTRANT','SORTANT')),
    sens varchar(20) not null
        check (sens in ('COMPANY_TO_CONTRACTOR','CONTRACTOR_TO_COMPANY')),

    projet_id bigint not null references projets(id),
    commande_id bigint references commandes(id),

    emetteur_id bigint references utilisateurs(id),
    destinataire_nom varchar(255),

    date_emission date,
    date_reception date,
    date_limite_reponse date,

    statut varchar(30) default 'Brouillon'
        check (statut in (
            'Brouillon','Émis','Reçu','En attente','Clôturé'
        )),

    objet text,
    commentaire text,

    genere_automatiquement boolean default false,
    date_creation timestamptz default now(),
    date_modification timestamptz default now(),
    cree_par bigint references utilisateurs(id)
);

create table transmittal_documents (
    id bigint generated always as identity primary key,
    transmittal_id bigint not null references transmittals(id) on delete cascade,
    document_mdr_id bigint not null references documents_mdr(id),
    revision_id bigint references document_revisions(id),
    commentaire text
);

    foreign key (transmittal_entrant_id)
    references transmittals(id) on delete set null;

alter table document_revisions
    add constraint fk_rev_transmittal_sortant
    foreign key (transmittal_sortant_id)
    references transmittals(id) on delete set null;

-- =========================================================
-- REVUE DOCUMENTAIRE
-- =========================================================

create table revues (
    id bigint generated always as identity primary key,
    document_mdr_id bigint not null references documents_mdr(id) on delete cascade,
    revision_id bigint references document_revisions(id),

    statut varchar(30) not null default 'En attente'
        check (statut in (
            'En attente','Ouverte','En cours',
            'En attente validation','Fermée','Annulée'
        )),

    date_ouverture date,
    date_limite date,
    date_fermeture date,

    code_retour_propose varchar(1)
        check (code_retour_propose in ('A','B','C','D')),
    code_retour_final varchar(1)
        check (code_retour_final in ('A','B','C','D')),

    -- acteurs
    doc_controller_id bigint references utilisateurs(id),
    leader_technique_id bigint references utilisateurs(id),
    gestionnaire_projet_id bigint references utilisateurs(id),

    commentaire_global text,
    date_creation timestamptz default now(),
    date_modification timestamptz default now(),
    cree_par bigint references utilisateurs(id)
);

create table assignations_revue (
    id bigint generated always as identity primary key,
    revue_id bigint not null references revues(id) on delete cascade,
    specialiste_id bigint not null references utilisateurs(id),

    statut varchar(20) default 'Assignée'
        check (statut in ('Assignée','En cours','Complétée','Refusée')),

    date_assignation timestamptz default now(),
    date_limite date,
    date_completion timestamptz
);

create table commentaires_revue (
    id bigint generated always as identity primary key,
    revue_id bigint not null references revues(id) on delete cascade,
    assignation_id bigint references assignations_revue(id),
    auteur_id bigint not null references utilisateurs(id),

    type_auteur varchar(20)
        check (type_auteur in (
            'SPECIALISTE','LEADER_TECHNIQUE',
            'GESTIONNAIRE','DOC_CONTROLLER'
        )),

    commentaire text not null,
    reference_document varchar(100),
    numero_commentaire int,

    statut varchar(20) default 'Ouvert'
        check (statut in ('Ouvert','Accepté','Rejeté','En attente')),

    reponse_contractor text,
    valide_par bigint references utilisateurs(id),

    date_creation timestamptz default now(),
    date_modification timestamptz default now()
);

-- =========================================================
-- MDDM — MASTER DOCUMENT DISTRIBUTION MATRIX
-- =========================================================

create table mddm (
    id bigint generated always as identity primary key,
    commande_id bigint not null references commandes(id) on delete cascade,

-- =========================================================
-- DOCUMENTS MDR — CMD-002 à CMD-015 (génération via DO)
-- =========================================================
do $$
declare
    cmd_code      text;
    cmd_list      text[] := array[
        'CMD-002-SAIP','CMD-003-SAIP','CMD-004-ABBC','CMD-005-SIMC',
        'CMD-006-TPWI','CMD-007-VLRC','CMD-008-SAIP','CMD-009-ABWI',
        'CMD-010-TPOE','CMD-011-ABOE','CMD-012-SIOE','CMD-013-SAOE',
        'CMD-014-TPHS','CMD-015-INHS'
    ];
    disc_codes    text[] := array[
        'ELE','INS','MEC','PIP','PRO',
        'CIV','STR','QUA','HSE','PJM'
    ];
    doc_type_codes text[] := array[
        'PID','SPC','DTS','NCT','LST',
        'RPT','PRO','PLN','SCH','MDR'
    ];
    classe_nums   int[]  := array[1,1,2,1,2,3,2,1,2,2];
    i             int;
    phase_code    text;
    mdr_id_val    bigint;
    doc_num       text;
    seq_map       text;
    statut_final  text;
    statut_cur    text;
    code_ret      text;
    rev_cur       text;
    est_fin       boolean;
    admin_id      bigint;
    c_id          bigint;
    c_projet_id   bigint;
    c_phase_id    bigint;
    c_pays_id     bigint;
    c_site_id     bigint;
    c_secteur_id  bigint;
    c_date_debut  date;
begin
    select id into admin_id
    from utilisateurs where email = 'admin@company.ga';

    foreach cmd_code in array cmd_list loop

        -- récupérer les infos de la commande
        select
            c.id, c.projet_id, c.phase_id,
            c.pays_id, c.site_id, c.secteur_id,
            c.date_debut_prevue,
            p.code
        into
            c_id, c_projet_id, c_phase_id,
            c_pays_id, c_site_id, c_secteur_id,
            c_date_debut,
            phase_code
        from commandes c
        join phases p on c.phase_id = p.id
        where c.code_commande = cmd_code;

        -- récupérer le mdr_id
        select m.id into mdr_id_val
        from mdr m
        where m.commande_id = c_id;

        -- SI le mdr_id_val est NULL, on ignore cette commande
        if mdr_id_val is null then
            continue;
        end if;

        for i in 1..10 loop

            -- déterminer séquence, statut final, statut courant
            if classe_nums[i] = 3 then
                seq_map      := 'IFI';
                statut_final := 'IFI';
                statut_cur   := 'IFI';
                code_ret     := case when i <= 6 then 'A' else null end;
                est_fin      := i <= 6;
                rev_cur      := case when i <= 6 then '01' else '00' end;

            elsif phase_code = 'EPC' and classe_nums[i] = 1 then
                seq_map      := 'IFR,IFA,IFC,MKP,ASB';
                statut_final := 'ASB';
                statut_cur   := case
                    when i <= 2 then 'IFR'
                    when i <= 5 then 'IFA'
                    else 'IFC' end;
                code_ret     := case when i <= 5 then 'B' else 'A' end;
                est_fin      := false;
                rev_cur      := case
                    when i <= 2 then '00'
                    when i <= 5 then '01'
                    else '02' end;

            elsif phase_code in ('BE-DE','BE') and classe_nums[i] = 1 then
                seq_map      := 'IFR,IFA,IFD';
                statut_final := 'IFD';
                statut_cur   := case
                    when i <= 3 then 'IFR'
                    when i <= 7 then 'IFA'
                    else 'IFD' end;
                code_ret     := case
                    when i <= 3 then 'C'
                    when i <= 7 then 'B'
                    else 'A' end;
                est_fin      := i > 7;
                rev_cur      := case
                    when i <= 3 then '01'
                    when i <= 7 then '02'
                    else '03' end;

            elsif phase_code in ('DE','PRO') and classe_nums[i] = 1 then
                seq_map      := 'IFR,IFA,IFC';
                statut_final := 'IFC';
                statut_cur   := case
                    when i <= 3 then 'IFR'
                    when i <= 6 then 'IFA'
                    else 'IFC' end;
                code_ret     := case
                    when i <= 3 then 'C'
                    when i <= 6 then 'B'
                    else 'A' end;
                est_fin      := i > 6;
                rev_cur      := case
                    when i <= 3 then '01'
                    when i <= 6 then '02'
                    else '03' end;

            else
                seq_map      := 'IFR,IFC';
                statut_final := 'IFC';
                statut_cur   := case when i <= 5 then 'IFR' else 'IFC' end;
                code_ret     := case when i <= 5 then 'B' else 'A' end;
                est_fin      := i > 5;
                rev_cur      := case when i <= 5 then '01' else '02' end;
            end if;

            -- construire le numéro de document
            doc_num := 'GA-MDJ-AGM-'
                || replace(cmd_code, 'CMD-', '') || '-'
                || disc_codes[i] || '-'
                || doc_type_codes[i] || '-'
                || lpad(i::text, 5, '0');

            insert into documents_mdr (
                mdr_id, commande_id, projet_id,
                numero_document, titre,
                phase_id, discipline_id, classe_num, doc_type_id,
                pays_id, site_id, secteur_id,
                duree_revue_company, duree_revue_contractor,
                statut_final_attendu, sequence_statuts,
                revision_courante, statut_courant, code_retour_courant,
                est_finalise,
                date_emission_prevue,
                date_emission_reelle,
                cree_par
            ) values (
                mdr_id_val,
                c_id,
                c_projet_id,
                doc_num,
                'Document ' || disc_codes[i] || ' - '
                    || doc_type_codes[i] || ' n°' || i
                    || ' [' || cmd_code || ']',
                c_phase_id,
                (select id from disciplines where code = disc_codes[i]),
                classe_nums[i],
                (select id from doc_types_technique
                 where code = doc_type_codes[i] limit 1),
                c_pays_id,
                c_site_id,
                c_secteur_id,
                15, 15,
                statut_final,
                seq_map,
                rev_cur,
                statut_cur,
                code_ret,
                est_fin,
                (c_date_debut + (i * 15 * interval '1 day'))::date,
                (c_date_debut + ((i * 15 + 5) * interval '1 day'))::date,
                admin_id
            );

        end loop;
    end loop;
end;
$$;
